# Deitar com parceiro ou parente (integrado do Bed Relations).
# As variantes por sexo do original viraram uma interação por relação; o evento
# pcr_bed.0001 escolhe o texto e a gravidez pelo sexo dos dois.
# Sem cooldown. A interação de prisioneiro do original ficou de fora: o Polycarater
# já tem pcr_impregnate_interaction.

# Cônjuge
pcr_bed_spouse_interaction = {
	category = interaction_category_friendly
	use_diplomatic_range = yes
	desc = pcr_bed_partner_interaction_desc

	is_shown = {
		scope:actor = { is_adult = yes }
		scope:recipient = {
			is_spouse_of = scope:actor
			is_adult = yes
			is_imprisoned = no
		}
	}

	is_valid_showing_failures_only = {
		scope:recipient = { is_available = yes }
	}

	auto_accept = yes

	on_accept = {
		pcr_bed_effect = { OPINION = 25 LOVER = 20 SOULMATE = 5 INCEST = no }
	}

	ai_frequency = 0
	ai_will_do = { base = 0 }
}
