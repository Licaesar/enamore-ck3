# Interações de prisão do Polycarater

# Casar à força um prisioneiro com alguém da corte do captor
pcr_force_marriage_interaction = {
	common_interaction = yes

	category = interaction_category_prison

	desc = pcr_force_marriage_interaction_desc

	interface = marriage
	send_options_exclusive = no

	redirect = {
		clear_saved_scope = secondary_actor
		scope:recipient = {
			save_scope_as = secondary_recipient
		}
	}

	# Candidatos do lado do captor: ele mesmo, os cortesãos e os filhos.
	# can_marry_trigger respeita o limite do traço pcr_unbound_unions.
	populate_actor_list = {
		scope:actor = {
			if = {
				limit = {
					sex_opposite_of = scope:recipient
					can_marry_trigger = yes
					# A fé permite este grau de parentesco
					OR = {
						NOT = { is_close_or_extended_family_of = scope:recipient }
						AND = {
							is_cousin_of = scope:recipient
							faith = { has_doctrine_parameter = allows_cousin_marriage }
						}
						AND = {
							OR = {
								is_uncle_or_aunt_of = scope:recipient
								is_nibling_of = scope:recipient
							}
							faith = { has_doctrine_parameter = allows_aunt_nephew_and_uncle_niece_marriage }
						}
						faith = { has_doctrine_parameter = allows_unrestricted_marriage }
					}
				}
				add_to_list = characters
			}
			every_courtier = {
				limit = {
					is_at_home = yes
					can_marry_trigger = yes
					sex_opposite_of = scope:recipient
					NOT = { is_spouse_of = scope:recipient }
					# A fé permite este grau de parentesco
					OR = {
						NOT = { is_close_or_extended_family_of = scope:recipient }
						AND = {
							is_cousin_of = scope:recipient
							faith = { has_doctrine_parameter = allows_cousin_marriage }
						}
						AND = {
							OR = {
								is_uncle_or_aunt_of = scope:recipient
								is_nibling_of = scope:recipient
							}
							faith = { has_doctrine_parameter = allows_aunt_nephew_and_uncle_niece_marriage }
						}
						faith = { has_doctrine_parameter = allows_unrestricted_marriage }
					}
				}
				add_to_list = characters
			}
			every_child = {
				limit = {
					NOT = { is_in_list = characters }
					can_marry_trigger = yes
					sex_opposite_of = scope:recipient
					NOT = { is_spouse_of = scope:recipient }
					# A fé permite este grau de parentesco
					OR = {
						NOT = { is_close_or_extended_family_of = scope:recipient }
						AND = {
							is_cousin_of = scope:recipient
							faith = { has_doctrine_parameter = allows_cousin_marriage }
						}
						AND = {
							OR = {
								is_uncle_or_aunt_of = scope:recipient
								is_nibling_of = scope:recipient
							}
							faith = { has_doctrine_parameter = allows_aunt_nephew_and_uncle_niece_marriage }
						}
						faith = { has_doctrine_parameter = allows_unrestricted_marriage }
					}
				}
				add_to_list = characters
			}
		}
	}

	populate_recipient_list = {
		scope:recipient = {
			add_to_list = characters
		}
	}

	send_option = {
		flag = matrilineal
		localization = MATRILINEAL_OFFER
		is_shown = {
			NAND = {
				scope:secondary_actor ?= { is_male = yes }
				scope:secondary_recipient ?= { is_male = yes }
			}
		}
		starts_enabled = {
			scope:secondary_actor ?= { is_female = yes }
		}
	}

	is_shown = {
		scope:recipient = {
			pcr_can_force_marry_trigger = yes
		}
	}

	on_accept = {
		pcr_force_marriage_effect = yes
	}

	auto_accept = yes

	ai_will_do = {
		base = -1000
	}
}

# Engravidar uma prisioneira; só para o jogador
pcr_impregnate_interaction = {
	category = interaction_category_prison
	icon = pregnant
	desc = pcr_impregnate_interaction_desc

	on_accept = {
		scope:recipient = { save_scope_as = target_captive }
		scope:actor = {
			trigger_event = {
				id = pcr_impregnate.0001
				days = 0
			}
		}
	}

	is_shown = {
		scope:actor = { is_ai = no }
		scope:recipient = {
			is_female = yes
			is_imprisoned_by = scope:actor
			is_adult = yes
		}
	}

	is_valid_showing_failures_only = {
		scope:actor = {
			is_male = yes
			fertility > 0
		}
		scope:recipient = {
			fertility > 0
			is_pregnant = no
		}
	}

	cooldown_against_recipient = { days = 30 }

	auto_accept = yes
}
