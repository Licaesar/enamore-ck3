# Costume das Uniões Livres. A busca de cônjuges do motor lê o limite da cultura
# antes dos gatilhos de script, por isso o limite precisa estar na tradição.
tradition_pcr_unbound_unions = {
	category = ritual

	layers = {
		0 = diplo
		1 = mena
		4 = temple.dds
	}

	can_pick = {
		custom_tooltip = {
			text = pcr_unbound_unions_tradition_requires_trait_tt
			scope:character = { has_trait = pcr_unbound_unions }
		}
		# Só uma tradição de casamento por cultura
		custom_tooltip = {
			text = only_one_marriage_tradition_allowed
			trigger_if = {
				limit = {
					OR = {
						NOT = { exists = scope:replacing }
						NOR = {
							scope:replacing = culture_tradition:tradition_monogamous
							scope:replacing = culture_tradition:tradition_polygamous
							scope:replacing = culture_tradition:tradition_concubines
						}
					}
				}
				NOR = {
					culture_tradition:tradition_monogamous = { is_in_list = traits }
					culture_tradition:tradition_polygamous = { is_in_list = traits }
					culture_tradition:tradition_concubines = { is_in_list = traits }
				}
			}
		}
	}

	can_pick_for_hybridization = {
		scope:character = { has_trait = pcr_unbound_unions }
		NOR = {
			culture_tradition:tradition_monogamous = { is_in_list = traits }
			culture_tradition:tradition_polygamous = { is_in_list = traits }
			culture_tradition:tradition_concubines = { is_in_list = traits }
		}
	}

	# Na 1.20 o limite de cônjuges fica em partners (substitui o da fé). Só esposas:
	# sem number_of_consorts, as concubinas seguem a fé.
	partners = {
		number_of_spouses = 10
	}

	ai_will_do = { value = 0 }
}
