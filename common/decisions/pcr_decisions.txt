# Estabelece o costume Enamore na cultura do governante com o traço
pcr_establish_enamore_decision = {
	decision_group_type = major

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}

	desc = pcr_establish_enamore_decision_desc
	selection_tooltip = pcr_establish_enamore_decision_tooltip
	confirm_text = pcr_establish_enamore_decision_confirm

	ai_check_interval = 0

	is_shown = {
		is_ruler = yes
		has_trait = pcr_enamore
		culture = {
			NOT = { has_cultural_tradition = tradition_pcr_enamore_customs }
		}
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_imprisoned = no
	}

	# Troca a tradição de casamento atual da cultura pela Enamore
	effect = {
		culture = {
			if = {
				limit = { has_cultural_tradition = tradition_monogamous }
				remove_culture_tradition = tradition_monogamous
			}
			if = {
				limit = { has_cultural_tradition = tradition_polygamous }
				remove_culture_tradition = tradition_polygamous
			}
			if = {
				limit = { has_cultural_tradition = tradition_concubines }
				remove_culture_tradition = tradition_concubines
			}
			add_culture_tradition = tradition_pcr_enamore_customs
		}
	}

	ai_will_do = { base = 0 }
}

# O jogador, numa cultura com o costume Enamore, compra o traço com piedade
pcr_adopt_enamore_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}

	desc = pcr_adopt_enamore_decision_desc
	selection_tooltip = pcr_adopt_enamore_decision_tooltip
	confirm_text = pcr_adopt_enamore_decision_confirm

	ai_check_interval = 0

	is_shown = {
		is_ai = no
		NOT = { has_trait = pcr_enamore }
		culture = {
			has_cultural_tradition = tradition_pcr_enamore_customs
		}
	}

	is_valid_showing_failures_only = {
		is_adult = yes
		is_imprisoned = no
	}

	cost = {
		piety = 500
	}

	effect = {
		add_trait = pcr_enamore
	}

	ai_will_do = { base = 0 }
}
