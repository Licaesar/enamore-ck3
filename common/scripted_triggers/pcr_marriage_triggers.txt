# Prisioneiro que pode ser casado à força pelo captor (escopo: o prisioneiro)
pcr_can_force_marry_trigger = {
	is_imprisoned_by = scope:actor
	NOT = { is_spouse_of = scope:actor }
	is_adult = yes
	NOT = { has_trait = eunuch_1 }
	NOT = { has_trait = beardless_eunuch }
	NOT = { has_trait = order_member }
}

# O genitor é ou foi concubina (ou concubino) de CHARACTER, mesmo morto (escopo: o genitor)
pcr_was_concubine_of_trigger = {
	OR = {
		is_concubine_of = $CHARACTER$
		any_former_concubinist = {
			even_if_dead = yes
			this = $CHARACTER$
		}
	}
}
