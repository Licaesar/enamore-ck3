# Sobrescritas de gatilhos vanilla de common/scripted_triggers/00_marriage_triggers.txt.
# O prefixo zz_ faz este arquivo carregar depois do vanilla. Os nomes não levam pcr_
# porque precisam ser iguais aos do vanilla para substituí-los.
# Regra: quem tem pcr_unbound_unions tem até 2 cônjuges e 3 concubinas; numa cultura com
# tradition_pcr_unbound_unions, quem não tem o traço fica com 1 cônjuge e sem concubinas.
# Sem o traço e sem a tradição, vale o comportamento vanilla.

can_marry_trigger = {
	can_marry_common_trigger = yes

	trigger_if = {
		limit = { has_trait = pcr_unbound_unions }
		custom_description = {
			text = pcr_unbound_unions_spouse_cap_tt
			any_spouse = { count < 2 }
		}
	}
	trigger_else_if = {
		limit = {
			culture = {
				has_cultural_tradition = tradition_pcr_unbound_unions
			}
		}
		custom_description = {
			text = pcr_unbound_unions_trait_required_for_secondary_spouse_tt
			any_spouse = { count < 1 }
		}
	}
	trigger_else = {
		allowed_more_spouses = yes
	}
	# Para casar com a própria concubina, use can_marry_character_trigger, como no vanilla
}

# Na 1.20 o vanilla usa scope:puppet_or_actor (o fantoche ou o próprio ator)
make_concubine_is_shown_trigger = {
	scope:puppet_or_actor = {
		trigger_if = {
			limit = {
				culture = {
					has_cultural_tradition = tradition_pcr_unbound_unions
				}
			}
			has_trait = pcr_unbound_unions
		}
		trigger_else = {
			allowed_concubines = yes
		}
		is_adult = yes
	}
	scope:recipient = {
		OR = {
			is_courtier_of = scope:puppet_or_actor
			is_imprisoned_by = scope:puppet_or_actor
		}
		can_become_concubine_of_character_valid_trigger = { CHARACTER = scope:puppet_or_actor }
	}
}

make_concubine_is_valid_trigger = {
	trigger_if = {
		limit = {
			scope:puppet_or_actor = {
				culture = {
					has_cultural_tradition = tradition_pcr_unbound_unions
				}
			}
		}
		custom_description = {
			text = pcr_unbound_unions_concubine_cap_tt
			object = scope:recipient
			scope:puppet_or_actor = {
				has_trait = pcr_unbound_unions
				number_of_concubines < 3
			}
		}
	}
	trigger_else = {
		custom_description = {
			text = take_concubine_allowed_more
			object = scope:recipient
			scope:puppet_or_actor = { allowed_more_concubines = yes }
		}
	}

	scope:recipient = {
		custom_tooltip = {
			text = take_concubine_ex_spouse
			NOT = {
				has_opinion_modifier = {
					modifier = divorced_me_opinion
					target = scope:puppet_or_actor
				}
			}
		}
		OR = {
			AND = {
				is_imprisoned_by = scope:puppet_or_actor
				is_married = yes
			}
			is_married = no
		}
		is_hostage = no
	}
}

# A janela "Encontrar concubina" usa este gatilho para montar a lista; numa cultura com
# a tradição, quem não tem o traço não recebe candidatas.
can_become_concubine_of_character_trigger = {
	$CHARACTER$ = {
		trigger_if = {
			limit = {
				culture = {
					has_cultural_tradition = tradition_pcr_unbound_unions
				}
			}
			has_trait = pcr_unbound_unions
		}
	}

	is_available_quick = {
		adult = yes
		ruler = no
		advanced_ruler = no
		hostage = no
	}
	trigger_if = {
		limit = { NOT = { is_imprisoned_by = $CHARACTER$ } }
		is_married = no
		is_betrothed = no
		is_concubine = no
		number_of_concubines = 0
	}
	trigger_if = {
		limit = {
			is_imprisoned_by = $CHARACTER$
			is_married = yes
		}
		NOT = { is_spouse_of = $CHARACTER$ }
	}
	NOR = {
		is_concubine_of = $CHARACTER$
		has_character_flag = has_been_offered_as_concubine
	}
	could_marry_character_trigger = { CHARACTER = $CHARACTER$ }
}

can_become_concubine_of_character_valid_trigger = {

			limit = {
				culture = {
					has_cultural_tradition = tradition_pcr_unbound_unions
				}
			}
			has_trait = pcr_unbound_unions
		}
	}

	is_available_quick = {
		adult = yes
		ruler = no
		advanced_ruler = no
	}
	trigger_if = {
		limit = { NOT = { is_imprisoned_by = $CHARACTER$ } }
		is_married = no
		is_betrothed = no
		is_concubine = no
		is_diarch = no
		number_of_concubines = 0
	}
	trigger_if = {
		limit = {
			is_imprisoned_by = $CHARACTER$
			is_married = yes
		}
		NOT = { is_spouse_of = $CHARACTER$ }
	}
	NOR = {
		is_concubine_of = $CHARACTER$

	can_take_as_concubine_character_trigger = { CHARACTER = $CHARACTER$ }
	has_any_disease_trigger = no
	has_easily_mocked_physical_attribute_trigger = no
}

can_be_offered_as_concubine_to_character_trigger = {
	$CHARACTER$ = {
		trigger_if = {
			limit = {
				culture = {
					has_cultural_tradition = tradition_pcr_unbound_unions
				}
			}
			has_trait = pcr_unbound_unions
		}
	}

	is_available_quick = {
		adult = yes
		ruler = no
		advanced_ruler = no
	}
	trigger_if = {
		limit = { NOT = { is_imprisoned_by = $GIVER$ } }
		is_married = no
		is_betrothed = no
		is_diarch = no
		OR = {
			is_concubine = no
			is_concubine_of = $GIVER$
		}
		number_of_concubines = 0
	}
	trigger_if = {
		limit = {
			is_imprisoned_by = $GIVER$
			is_married = yes
		}
		NOR = {
			is_spouse_of = $CHARACTER$
			is_spouse_of = $GIVER$
		}
	}
	NOR = {
		is_concubine_of = $CHARACTER$
		has_character_flag = has_been_offered_as_concubine
	}
	NOT = { is_theocratic_lessee = yes }
	could_marry_character_trigger = { CHARACTER = $CHARACTER$ }
}
