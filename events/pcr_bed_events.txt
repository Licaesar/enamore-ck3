namespace = pcr_bed

# Deitar com parceiro ou parente. Root: o ator. Escopos de pcr_bed_effect:
# pcr_bed_partner (o recipiente) e, entre sexos opostos, pcr_bed_mother e pcr_bed_father.
pcr_bed.0001 = {
	type = character_event
	title = pcr_bed.0001.t
	desc = {
		first_valid = {
			# Ator mulher com homem
			triggered_desc = {
				trigger = {
					exists = scope:pcr_bed_mother
					scope:pcr_bed_mother ?= root
				}
				desc = pcr_bed.0001.desc_receive
			}
			# Ator homem com mulher
			triggered_desc = {
				trigger = { exists = scope:pcr_bed_mother }
				desc = pcr_bed.0001.desc_give
			}
			# Dois homens
			triggered_desc = {
				trigger = { is_female = no }
				desc = pcr_bed.0001.desc_male
			}
			# Duas mulheres
			desc = pcr_bed.0001.desc_female
		}
	}
	theme = lover_relation

	left_portrait = {
		character = root
		animation = love
	}
	right_portrait = {
		character = scope:pcr_bed_partner
		animation = love
	}

	immediate = {
		pcr_orgy_undress_effect = yes
		scope:pcr_bed_partner = { pcr_orgy_undress_effect = yes }
	}

	# Terminar dentro: um filho
	option = {
		name = {
			trigger = { scope:pcr_bed_father ?= root }
			text = pcr_bed.0001.a_give
		}
		name = {
			trigger = { scope:pcr_bed_mother ?= root }
			text = pcr_bed.0001.a_receive
		}
		trigger = {
			exists = scope:pcr_bed_mother
			scope:pcr_bed_mother = { pcr_bed_can_conceive_trigger = yes }
		}
		scope:pcr_bed_mother = { pcr_bed_conceive_effect = { CHILDREN = 1 } }
		trigger_event = pcr_bed.0002
	}

	# Terminar dentro: gêmeos
	option = {
		name = {
			trigger = { scope:pcr_bed_father ?= root }
			text = pcr_bed.0001.twins_give
		}
		name = {
			trigger = { scope:pcr_bed_mother ?= root }
			text = pcr_bed.0001.twins_receive
		}
		trigger = {
			exists = scope:pcr_bed_mother
			scope:pcr_bed_mother = { pcr_bed_can_conceive_trigger = yes }
		}
		scope:pcr_bed_mother = { pcr_bed_conceive_effect = { CHILDREN = 2 } }
		trigger_event = pcr_bed.0002
	}

	# Terminar dentro sem gravidez possível (já grávida ou sem fertilidade)
	option = {
		name = {
			trigger = { scope:pcr_bed_father ?= root }
			text = pcr_bed.0001.a_give
		}
		name = {
			trigger = { scope:pcr_bed_mother ?= root }
			text = pcr_bed.0001.a_receive
		}
		trigger = {
			exists = scope:pcr_bed_mother
			scope:pcr_bed_mother = { pcr_bed_can_conceive_trigger = no }
		}
		custom_tooltip = pcr_bed.0001.no_conception_tt
	}

	# Sexos opostos: tirar
	option = {
		name = {
			trigger = { scope:pcr_bed_father ?= root }
			text = pcr_bed.0001.b_give
		}
		name = {
			trigger = { scope:pcr_bed_mother ?= root }
			text = pcr_bed.0001.b_receive
		}
		trigger = { exists = scope:pcr_bed_mother }
	}

	# Mesmo sexo
	option = {
		name = {
			trigger = { is_female = no }
			text = pcr_bed.0001.c_male
		}
		name = {
			trigger = { is_female = yes }
			text = pcr_bed.0001.c_female
		}
		trigger = { NOT = { exists = scope:pcr_bed_mother } }
	}

	after = {
		pcr_orgy_dress_effect = yes
		scope:pcr_bed_partner = { pcr_orgy_dress_effect = yes }
	}
}

# Escolha do sexo da criança, logo depois da concepção. Root: o ator.
# A escolha vale para todos os filhos da gravidez (um ou gêmeos).
pcr_bed.0002 = {
	type = character_event
	title = pcr_bed.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:pcr_bed_mother = { var:pcr_bed_children ?= 2 } }
				desc = pcr_bed.0002.desc_twins
			}
			desc = pcr_bed.0002.desc
		}
	}
	theme = lover_relation

	left_portrait = {
		character = root
		animation = happiness
	}
	right_portrait = {
		character = scope:pcr_bed_partner
		animation = love
	}

	trigger = {
		scope:pcr_bed_mother = { has_variable = pcr_bed_children }
	}

	option = {
		name = {
			trigger = { scope:pcr_bed_mother = { var:pcr_bed_children ?= 2 } }
			text = pcr_bed.0002.a_twins
		}
		name = {
			trigger = { scope:pcr_bed_mother = { NOT = { var:pcr_bed_children ?= 2 } } }
			text = pcr_bed.0002.a
		}
		custom_tooltip = pcr_bed.0002.a_tt
		scope:pcr_bed_mother = { pcr_bed_set_gender_effect = { GENDER = male } }
	}

	option = {
		name = {
			trigger = { scope:pcr_bed_mother = { var:pcr_bed_children ?= 2 } }
			text = pcr_bed.0002.b_twins
		}
		name = {
			trigger = { scope:pcr_bed_mother = { NOT = { var:pcr_bed_children ?= 2 } } }
			text = pcr_bed.0002.b
		}
		custom_tooltip = pcr_bed.0002.b_tt
		scope:pcr_bed_mother = { pcr_bed_set_gender_effect = { GENDER = female } }
	}

	# Deixa o sexo ao acaso
	option = {
		name = pcr_bed.0002.c
	}
}
