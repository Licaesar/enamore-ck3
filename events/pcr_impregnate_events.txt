namespace = pcr_impregnate

# Gravidez forçada de uma prisioneira
pcr_impregnate.0001 = {
	type = character_event
	title = pcr_impregnate.0001.t

	# Texto alternativo a partir da segunda vez, se a prisioneira já virou favorável
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = { exists = scope:pcr_first_time }
					scope:target_captive = { has_character_flag = pcr_forced_positive_relation }
				}
				desc = pcr_impregnate.0001.desc_pleasure
			}
			desc = pcr_impregnate.0001.desc
		}
	}

	theme = intrigue

	left_portrait = {
		character = root
		animation = scheme
	}

	right_portrait = {
		character = scope:target_captive

		# Favorável: vergonha
		triggered_animation = {
			trigger = { has_character_flag = pcr_forced_positive_relation }
			animation = shame
		}

		# Hostil: masmorra
		triggered_animation = {
			trigger = {
				has_opinion_modifier = {
					modifier = pcr_forced_union_negative_opinion
					target = root
				}
			}
			animation = prisondungeon
		}

		# Primeira vez: medo
		animation = fear
	}

	immediate = {
		# Marca a primeira vez
		if = {
			limit = { NOT = { scope:target_captive = { has_character_flag = pcr_impregnate_interacted } } }
			save_scope_value_as = { name = pcr_first_time value = yes }
			scope:target_captive = { add_character_flag = pcr_impregnate_interacted }
		}

		# Gravidez garantida. O flag allow_event_pregnancy impede que a regra de gravidez
		# realista do Love Marriage encerre a gravidez (prisioneira e pai não estão juntos);
		# sem esse mod, o flag não faz nada.
		if = {
			limit = { scope:target_captive = { is_pregnant = no } }
			scope:target_captive = {
				add_character_flag = { flag = allow_event_pregnancy days = 61 }
				make_pregnant = { father = root }
				add_character_flag = pcr_force_bred_by_player
			}
		}

		# Opinião: 80% hostil, 20% favorável; favorável fica permanente
		if = {
			limit = { NOT = { scope:target_captive = { has_character_flag = pcr_forced_positive_relation } } }
			random_list = {
				80 = {
					scope:target_captive = {
						remove_opinion = { modifier = pcr_forced_union_positive_opinion target = root }
						add_opinion = { modifier = pcr_forced_union_negative_opinion target = root }
					}
				}
				20 = {
					scope:target_captive = {
						remove_opinion = { modifier = pcr_forced_union_negative_opinion target = root }
						add_opinion = { modifier = pcr_forced_union_positive_opinion target = root }
						add_character_flag = pcr_forced_positive_relation
						random = { chance = 50 set_relation_lover = root }
					}
				}
			}
		}
	}

	option = { name = pcr_impregnate.0001.a }
}

# Nascimento do filho de uma prisioneira solteira: aviso ao jogador (oculto)
# Chamado por pcr_on_birth_child; root é o recém-nascido
pcr_impregnate.0002 = {
	type = character_event
	hidden = yes

	trigger = {
		exists = mother
		real_father ?= { is_ai = no }
	}

	immediate = {
		# Dentro de mother = { }, real_father seria o pai da mãe; por isso salva o da criança
		real_father = { save_scope_as = pcr_child_father }
		if = {
			limit = {
				mother = {
					OR = {
						has_character_flag = pcr_force_bred_by_player
						is_imprisoned_by = scope:pcr_child_father
					}
				}
			}

			mother = { remove_character_flag = pcr_force_bred_by_player }

			if = {
				limit = { mother = { is_married = no } }
				scope:pcr_child_father = {
					send_interface_toast = {
						type = event_toast_effect_neutral
						title = pcr_impregnate_birth_toast
						desc = pcr_impregnate_birth_toast_desc
					}
				}
			}
		}
	}
}
